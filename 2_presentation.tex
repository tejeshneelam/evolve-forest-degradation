            \begin{block}{Physics Hazard Formulation}
                \footnotesize
                $$P_{\text{landslide}} = \text{clip}(0.48 S_{\theta} + 0.32 C_{\text{decay}}\Omega(\theta) + 0.20 R_{90\text{d}}\Omega(\theta), 0.02, 0.98)$$

                    \item \textbf{Seasonal Multiplier ($M_{\text{season}}$):} $1.5\times$ for peak fire season (Feb--May in Western Ghats), $0.8\times$ dry months.
